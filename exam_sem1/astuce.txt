// Calculer et Afficher le nombre des recrues pour un poste donné
          (typeid pour obtenir le type réel de chaque candidat. Si le type correspond à celui de Recrue, nous effectuons un static_cast pour convertir le pointeur de Candidat en pointeur de Recrue)
      class application
int nombre_recrues_pour_poste(const std::string& poste) const {
        int count = 0;
        for (const Candidat* candidat : candidats) {
            if (typeid(*candidat) == typeid(Recrue)) {
                const Recrue* recrue = static_cast<const Recrue*>(candidat);
                if (recrue->get_poste() == poste) {
                    count++;
                }
            }
        }
        return count;
    }
